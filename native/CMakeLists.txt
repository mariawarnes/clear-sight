cmake_minimum_required(VERSION 3.25)
project(ClearSight VERSION 0.3.13 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(HC_BUILD_PLUGIN "Build the Skyrim plugin as well as GPU tests" ON)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
if(HC_BUILD_PLUGIN)
    set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
    set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
    set(COMMONLIB_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../build/CommonLibSSE-NG" CACHE PATH "Pinned CommonLib source")
    add_subdirectory("${COMMONLIB_SOURCE}" commonlib)
    target_compile_options(CommonLibSSE PRIVATE /MP4)
endif()
add_library(ClearSightTextures STATIC src/TextureCache.cpp src/GreyscaleShader.cpp third_party/dxc/DxilHash.cpp)
target_include_directories(ClearSightTextures PUBLIC src)
target_compile_definitions(ClearSightTextures PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(ClearSightTextures PUBLIC d3d11 d3dcompiler dxgi)
if(HC_BUILD_PLUGIN)
    add_library(ClearSight SHARED src/Plugin.cpp)
    target_link_libraries(ClearSight PRIVATE CommonLibSSE::CommonLibSSE ClearSightTextures)
    target_compile_options(ClearSight PRIVATE /Zc:preprocessor /bigobj /EHsc)
    install(TARGETS ClearSight RUNTIME DESTINATION SKSE/Plugins COMPONENT Native)
    install(FILES ClearSight.ini DESTINATION SKSE/Plugins COMPONENT Native)
endif()
add_executable(ClearSightTests tests/Tests.cpp)
target_link_libraries(ClearSightTests PRIVATE ClearSightTextures)
enable_testing()
add_test(NAME NativeRendering COMMAND ClearSightTests)
